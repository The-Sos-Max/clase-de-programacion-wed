import mongoose from "mongoose";
import Autor from "../models/Autor.js";

// POST /api/autores  → crear un autor
export async function crearAutor(req, res) {
  try {
    const autor = await Autor.create(req.body);
    res.status(201).json(autor);
  } catch (error) {
    res.status(400).json({ mensaje: error.message });
  }
}

// GET /api/autores  → listar todos los autores
export async function listarAutores(req, res) {
  try {
    const autores = await Autor.find().sort({ nombre: 1 });
    res.json(autores);
  } catch (error) {
    res.status(500).json({ mensaje: error.message });
  }
}

// GET /api/autores/:id  → consultar un autor por su id
export async function obtenerAutor(req, res) {
  try {
    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      return res.status(400).json({ mensaje: "ID inválido" });
    }

    const autor = await Autor.findById(id);
    if (!autor) {
      return res.status(404).json({ mensaje: "Autor no encontrado" });
    }

    res.json(autor);
  } catch (error) {
    res.status(500).json({ mensaje: error.message });
  }
}

import mongoose from "mongoose";
import Comic from "../models/Comic.js";
import Autor from "../models/Autor.js";

// POST /api/comics  → crear un cómic
export async function crearComic(req, res) {
  try {
    const { autor } = req.body;

    // Validamos que el autor referenciado exista
    if (!mongoose.isValidObjectId(autor) || !(await Autor.exists({ _id: autor }))) {
      return res.status(400).json({ mensaje: "El autor indicado no existe" });
    }

    const comic = await Comic.create(req.body);
    res.status(201).json(comic);
  } catch (error) {
    res.status(400).json({ mensaje: error.message });
  }
}

// GET /api/comics  → listar cómics (con filtros opcionales)
// Ejemplos: /api/comics?editorial=DC Comics   |   /api/comics?autor=<idAutor>
export async function listarComics(req, res) {
  try {
    const filtro = {};
    if (req.query.editorial) filtro.editorial = req.query.editorial;
    if (req.query.autor) filtro.autor = req.query.autor;
    if (req.query.titulo) {
  filtro.titulo = {
    $regex: req.query.titulo,
    $options: "i"
  };
}
    // populate() reemplaza el id del autor por el documento completo
    const comics = await Comic.find(filtro).populate("autor", "nombre nacionalidad");
    res.json(comics);
  } catch (error) {
    res.status(500).json({ mensaje: error.message });
  }
}

// GET /api/comics/:id  → consultar un cómic por su id
export async function obtenerComic(req, res) {
  try {
    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      return res.status(400).json({ mensaje: "ID inválido" });
    }

    const comic = await Comic.findById(id).populate("autor", "nombre nacionalidad");
    if (!comic) {
      return res.status(404).json({ mensaje: "Cómic no encontrado" });
    }

    res.json(comic);
  } catch (error) {
    res.status(500).json({ mensaje: error.message });
  }
}

 // PUT /api/comics/:id → actualizar un cómic
export async function actualizarComic(req, res) {
  try {
    const { id } = req.params;

    // Validar que el ID tenga un formato correcto
    if (!mongoose.isValidObjectId(id)) {
      return res.status(400).json({ mensaje: "ID inválido" });
    }

    // Si se envía un autor nuevo, comprobar que exista
    if (req.body.autor !== undefined) {
      if (
        !mongoose.isValidObjectId(req.body.autor) ||
        !(await Autor.exists({ _id: req.body.autor }))
      ) {
        return res.status(400).json({
          mensaje: "El autor indicado no existe"
        });
      }
    }

    const comic = await Comic.findByIdAndUpdate(
      id,
      req.body,
      {
        new: true,
        runValidators: true
      }
    ).populate("autor", "nombre nacionalidad");

    if (!comic) {
      return res.status(404).json({
        mensaje: "Cómic no encontrado"
      });
    }

    res.json(comic);
  } catch (error) {
    res.status(400).json({ mensaje: error.message });
  }
}

// DELETE /api/comics/:id → eliminar un cómic
export async function eliminarComic(req, res) {
  try {
    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      return res.status(400).json({ mensaje: "ID inválido" });
    }

    const comic = await Comic.findByIdAndDelete(id);

    if (!comic) {
      return res.status(404).json({
        mensaje: "Cómic no encontrado"
      });
    }

    res.json({
      mensaje: "Cómic eliminado correctamente",
      comic
    });
  } catch (error) {
    res.status(500).json({ mensaje: error.message });
  }
}

import "dotenv/config";
import { readFile } from "node:fs/promises";
import mongoose from "mongoose";
import Autor from "./models/Autor.js";
import Comic from "./models/Comic.js";

// Lee un archivo JSON relativo a este archivo
async function leerJSON(ruta) {
  const contenido = await readFile(new URL(ruta, import.meta.url), "utf-8");
  return JSON.parse(contenido);
}

async function seed() {
  await mongoose.connect(process.env.MONGODB_URI);
  console.log("Conectado a MongoDB Atlas");

  const autoresData = await leerJSON("../data/autores.json");
  const comicsData = await leerJSON("../data/comics.json");

  // Borra lo que haya, para poder ejecutar el seed varias veces
  await Comic.deleteMany();
  await Autor.deleteMany();

  // 1. Insertar autores
  const autores = await Autor.insertMany(autoresData);

  // 2. Diccionario nombre → _id
  const idPorNombre = Object.fromEntries(autores.map((a) => [a.nombre, a._id]));

  // 3. Reemplazar autorNombre por la referencia real
  const comics = comicsData.map(({ autorNombre, ...resto }) => ({
    ...resto,
    autor: idPorNombre[autorNombre]
  }));
  await Comic.insertMany(comics);

  console.log(`Insertados ${autores.length} autores y ${comics.length} cómics`);
  await mongoose.disconnect();
}

seed().catch((error) => {
  console.error(" Error en el seed:", error.message);
  process.exit(1);
});

// Como el cliente lo sirve la misma API, basta con una ruta relativa
const API_URL = "/api";

// Función base: hace la petición, convierte la respuesta a JSON y lanza un error si algo falla
async function peticion(ruta, opciones = {}) {
  const respuesta = await fetch(`${API_URL}${ruta}`, {
    headers: { "Content-Type": "application/json" },
    ...opciones
  });

  const datos = await respuesta.json();

  // respuesta.ok es false cuando el código HTTP es 400, 404, 500...
  if (!respuesta.ok) {
    throw new Error(datos.mensaje || "Error en la petición");
  }

  return datos;
}

// ---------- Autores ----------

// GET /api/autores
export function listarAutores() {
  return peticion("/autores");
}

// POST /api/autores
export function crearAutor(autor) {
  return peticion("/autores", {
    method: "POST",
    body: JSON.stringify(autor)
  });
}

// ---------- Cómics ----------

// GET /api/comics  o  GET /api/comics?autor=...&editorial=...
export function listarComics(filtros = {}) {
  const params = new URLSearchParams();

  // Solo agregamos los filtros que tengan valor
  Object.entries(filtros).forEach(([clave, valor]) => {
    if (valor) params.set(clave, valor);
  });

  const query = params.toString();
  return peticion(`/comics${query ? `?${query}` : ""}`);
}

// POST /api/comics
export function crearComic(comic) {

  });

// Llena un <select> con opciones, conservando la selección actual si aún existe
function llenarSelect(select, opciones, textoInicial) {
  const seleccionado = select.value;

  select.replaceChildren(new Option(textoInicial, ""));
  opciones.forEach(({ valor, texto }) => select.add(new Option(texto, valor)));

  select.value = seleccionado;
  if (select.selectedIndex === -1) select.selectedIndex = 0;
}

function renderComics(comics) {
  listaComics.replaceChildren();

  totalComics = comics.length;

  const totalPaginas = Math.max(
    1,
    Math.ceil(totalComics / comicsPorPagina)
  );

  if (pagina > totalPaginas) {
    pagina = totalPaginas;
  }

  contador.textContent =
    totalComics === 1
      ? "1 cómic encontrado"
      : `${totalComics} cómics encontrados`;

  paginaActual.textContent = `Página ${pagina} de ${totalPaginas}`;

  btnAnterior.disabled = pagina === 1;
  btnSiguiente.disabled = pagina === totalPaginas;

  if (totalComics === 0) {
    listaComics.append(
      crearElemento("p", "No se encontraron cómics con ese filtro.", "vacio")
    );
    return;
  }

  const inicio = (pagina - 1) * comicsPorPagina;
  const comicsPagina = comics.slice(inicio, inicio + comicsPorPagina);

  comicsPagina.forEach((comic) => {
    const tarjeta = document.createElement("article");
    tarjeta.className = "comic";

    tarjeta.append(
      crearElemento("h3", comic.titulo),
      crearElemento("p", `✍️ ${comic.autor?.nombre ?? "Autor desconocido"}`),
      crearElemento(
        "p",
        [comic.editorial, comic.anioPublicacion].filter(Boolean).join(" · ")
      ),
      crearElemento("p", formatoPeso.format(comic.precio), "precio"),
      crearElemento("p", `Stock: ${comic.stock}`)
    );

    // Botón para editar
    const btnEditar = crearElemento("button", "Editar");
    btnEditar.type = "button";

    btnEditar.addEventListener("click", async () => {
      await editarComic(comic);
    });

    // Botón para eliminar
    const btnEliminar = crearElemento("button", "Eliminar");
    btnEliminar.type = "button";
    btnEliminar.className = "secundario";

    btnEliminar.addEventListener("click", async () => {
      await borrarComic(comic);
    });

    // Contenedor de los botones
    const acciones = document.createElement("div");
    acciones.className = "acciones";
    acciones.append(btnEditar, btnEliminar);

    tarjeta.append(acciones);
    listaComics.append(tarjeta);
  });
}
async function borrarComic(comic) {
  const confirmar = confirm(
    `¿Seguro que deseas eliminar "${comic.titulo}"?`
  );

  if (!confirmar) return;

  try {
    await eliminarComic(comic._id);

    mostrarMensaje(`Cómic "${comic.titulo}" eliminado correctamente`);

    await Promise.all([
      cargarEditoriales(),
      buscar()
    ]);
  } catch (error) {
    mostrarMensaje(error.message, "error");
  }
}

// La API no tiene un endpoint de editoriales, así que las sacamos de los cómics
async function cargarEditoriales() {
  const comics = await listarComics();

  llenarSelect(
    filtroEditorial,
    editoriales.map((editorial) => ({ valor: editorial, texto: editorial })),

  );

// Consulta los cómics aplicando los filtros elegidos y los dibuja
async function buscar() {
   pagina = 1;
  const comics = await listarComics({
    titulo: filtroTitulo.value.trim(),
    autor: filtroAutor.value,
    editorial: filtroEditorial.value
  });

  renderComics(comics);
}

async function editarComic(comic) {
  const titulo = prompt("Título del cómic:", comic.titulo);
  if (titulo === null) return;

  const precioTexto = prompt(
    "Precio en COP:",
    String(comic.precio)
  );
  if (precioTexto === null) return;

  const stockTexto = prompt(
    "Cantidad en stock:",
    String(comic.stock ?? 0)
  );
  if (stockTexto === null) return;

  // Convertimos los valores a sus tipos correctos
  const precio = Number(precioTexto);
  const stock = Number(stockTexto);

  // Validamos el título
  if (!titulo.trim()) {
    mostrarMensaje("El título es obligatorio", "error");
    return;
  }

  // Validamos el precio
  if (!Number.isFinite(precio) || precio <= 0) {
    mostrarMensaje("El precio debe ser mayor que cero", "error");
    return;
  }

  // Validamos el stock
  if (!Number.isInteger(stock) || stock < 0) {
    mostrarMensaje(
      "El stock debe ser un entero mayor o igual a cero",
      "error"
    );
    return;
  }

  try {
    await actualizarComic(comic._id, {
      titulo: titulo.trim(),
      precio,
      stock
    });

    mostrarMensaje("Cómic actualizado correctamente");
    await buscar();
  } catch (error) {
    mostrarMensaje(error.message, "error");
  }
}

async function borrarComic(comic) {
  // Pedimos confirmación antes de eliminar
  const confirmar = confirm(
    `¿Seguro que deseas eliminar "${comic.titulo}"?`
  );

  // Si el usuario cancela, no hacemos nada
  if (!confirmar) return;

  try {
    // Enviamos la petición DELETE a la API
    await eliminarComic(comic._id);

    // Mostramos el mensaje de éxito
    mostrarMensaje("Cómic eliminado correctamente");

    // Actualizamos la lista de cómics
    await buscar();
  } catch (error) {
    mostrarMensaje(error.message, "error");
  }
}

async function editarComic(comic) {
  const titulo = prompt("Nuevo título:", comic.titulo);
  if (titulo === null) return;

  const precioTexto = prompt("Nuevo precio:", comic.precio);
  if (precioTexto === null) return;

  const editorial = prompt("Nueva editorial:", comic.editorial || "");
  if (editorial === null) return;

  const stockTexto = prompt("Nuevo stock:", comic.stock);
  if (stockTexto === null) return;

  const precio = Number(precioTexto);
  const stock = Number(stockTexto);

  if (
    !titulo.trim() ||
    precioTexto.trim() === "" ||
    !Number.isFinite(precio) ||
    precio < 0 ||
    stockTexto.trim() === "" ||
    !Number.isInteger(stock) ||
    stock < 0
  ) {
    mostrarMensaje(
      "Verifica el título, el precio y el stock.",
      "error"
    );
    return;
  }

  try {
    await actualizarComic(comic._id, {
      titulo: titulo.trim(),
      precio,
      editorial: editorial.trim(),
      stock
    });

    mostrarMensaje(`Cómic "${titulo}" actualizado correctamente`);

    await buscar();
  } catch (error) {
    mostrarMensaje(error.message, "error");
  }
}
